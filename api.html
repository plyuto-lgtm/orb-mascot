React

import { Orb } from 'orb-mascot/react';

<Orb identity={agent.id} preset="avatar" size="lg" status={agent.status} />
<Orb body="bear" color="#1E6DF6" shade="gradient" ref={orb}
     onClick={() => orb.current.run('surprise')} />

Without React

import Mascot from 'orb-mascot';   // or <script src="mascot.js">

const m = new Mascot(svg, { body: 'bear', color: '#1E6DF6' });
m.start(); await m.run('think');
